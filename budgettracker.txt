#!/bin/bash
echo "=== Budget Tracker (Bash) ==="
echo -n "Enter you monthly income (numbers only): "
read income

total_expenses=0
expense_count=0

echo "Enter expenses one at a time. Type done to finish."
while true; do
        echo -n "Enter an expense amount (or done): "
        read expense
if [[ "$expense" == "done" ]]; then
        break
fi
total_expenses=$(( total_expenses + expense ))
expense_count=$(( expense_count + 1 ))
done

balance=$(( income - total_expenses ))
echo ""
echo "=== Summary ==="
echo "Income: $income"
echo "Total Expenses: $total_expenses"
echo "Expense Entries: $expense_count"
echo "Remaining: $balance"
